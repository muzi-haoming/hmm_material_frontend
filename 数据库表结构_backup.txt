-- 确认和完成状态的公共触发器函数
create or replace function check_bill_status_rules()
returns trigger as $$
declare
	v_allowed_cols text[] := null;
	r record;             -- 存放查出来的外键信息
    v_sql text;           -- 存放动态生成的 sql
    v_has_ref boolean;    -- 检查结果
    v_ref_table_name text; -- 报错用的表名
begin
	if (new.is_confirmed <> old.is_confirmed) and (new.is_completed <> old.is_completed) then
		raise exception '非法操作：禁止同时修改[确认状态]和[完成状态]! 请分步操作。';
	end if;

    -- 禁止跳过确认直接完成
    if (old.is_confirmed = false and new.is_completed = true) then
        raise exception '非法操作：单据必须先确认后才能完成!';
    end if;

	-- 1. 设置和取消完成状态, 如果没有夹带私货, 直接通过
	if (old.is_confirmed and new.is_confirmed and (new.is_completed <> old.is_completed)) then
		v_allowed_cols := array['modifier', 'modify_time', 'is_completed'];
		if (hstore(new) - v_allowed_cols) <> (hstore(old) - v_allowed_cols) then
			raise exception '修改完成状态时存在修改其他数据的行为! (差异字段: %)', (hstore(new) - v_allowed_cols) - (hstore(OLD) - v_allowed_cols);
		end if;
		return new;
	end if;
	
	-- 2. 设置和取消确认状态, 如果没有夹带私货, 直接通过
	if (not old.is_completed and not new.is_completed and (new.is_confirmed <> old.is_confirmed)) then
		v_allowed_cols := array['modifier', 'modify_time', 'is_confirmed'];
		if (hstore(new) - v_allowed_cols) <> (hstore(old) - v_allowed_cols) then
			raise exception '修改确认状态时存在修改其他数据的行为! (差异字段: %)', (hstore(new) - v_allowed_cols) - (hstore(old) - v_allowed_cols);
		end if;
		-- 取消确认校验逻辑
		if (new.is_confirmed = false) then
            -- 循环遍历所有指向当前表的外键
            -- conrelid: 引用表(子表), confrelid: 被引用表(父表/当前表)
            for r in 
                select 
					conrelid::regclass::text as table_name, -- 获取子表表名
                    a.attname as col_name,                  -- 获取子表外键列名
                    obj_description(conrelid) as table_comment -- 获取子表的注释(如果有)
                from pg_constraint c
                join pg_attribute a on a.attrelid = c.conrelid and a.attnum = any(c.conkey)
                where 
					c.confrelid = tg_relid -- tg_relid 是当前触发器所在的表 oid
					and c.contype = 'f'        -- 只找外键 (foreign key)
            loop
                -- 构造动态查询语句: select true from 子表 where 外键列 = old.id limit 1
                v_sql := format('select true from %s where %I = $1 limit 1', r.table_name, r.col_name);
                -- 执行动态 sql
                execute v_sql using old.id into v_has_ref;
                -- 如果找到了关联数据
                if v_has_ref then
                    -- 优先使用表注释作为报错名称，如果没有注释则用表名
                    v_ref_table_name := coalesce(r.table_comment, r.table_name);
                    raise exception '无法取消确认! 存在其他业务数据: [%] (表: %)', v_ref_table_name, r.table_name;
                end if;
            end loop;
        end if;
		return new;
	end if;
	
	-- 合法的修改操作, 直接通过
	if (not old.is_confirmed and not new.is_confirmed and not old.is_completed and not new.is_completed) then
		return new;
	end if;

	raise exception '未知的非法操作!'
end;
$$ language plpgsql set search_path = public, pg_temp;

-- 私用触发器函数
-- 创建/更新反向同步函数 (兼容 INSERT 和 UPDATE)
create or replace function sync_profile_role_change()
returns trigger as $$
declare
    v_role_code varchar;
begin
    -- 逻辑判断：
    -- 1. 如果是 INSERT (新增档案)，直接同步
    -- 2. 如果是 UPDATE (修改档案)，只有 role_id 变了才同步
    if (tg_op = 'INSERT') or (tg_op = 'UPDATE' and old.role_id is distinct from new.role_id) then
        
        -- 查出新的 role_code
        select code into v_role_code from "role" where id = new.role_id;
        
        -- 防御性检查：如果查不到角色（比如外键还没提交），则不执行
        if v_role_code is not null then
            -- 更新 supabase auth 元数据
            update auth.users
            set raw_app_meta_data = 
                coalesce(raw_app_meta_data, '{}'::jsonb) || -- 确保不为 null
                jsonb_build_object(
                    'role_code', v_role_code, 
                    'role_id', new.role_id
                )
            where id = new.id;
        end if;
    end if;
    return new;
end;
$$ language plpgsql set search_path = public;

-- 私用触发器函数
-- 从 auth.users 同步用户信息到 profile
create or replace function sync_user_info()
returns trigger as $$
declare
    default_role_id uuid;
    default_role_code text := 'guest';
begin
    -- 1. 获取默认角色 ID
    select id into default_role_id from public.role where code = default_role_code limit 1;
    -- 2. 同步到业务表
    insert into public.profile (id, role_id)
    values (new.id, default_role_id);
    -- 3. 【关键】同步到 Auth 元数据，确保 JWT 立即生效
    update auth.users
    set raw_app_meta_data = raw_app_meta_data || jsonb_build_object('role_code', default_role_code, 'role_id', default_role_id)
    where id = new.id;
    return new;
end;
$$ language plpgsql security definer set search_path = public, auth;
-- 删除 profile 数据的时候同步删除对应的头像
create or replace function delete_avatar()
returns trigger as $$
begin
	delete from storage.objects where bucket_id = 'avatar' and name like (old.id::text || '.%');
    return old;
end;
$$ language plpgsql set search_path = public, storage;

-- 存储桶
insert into storage.buckets (id, name, public) values ('avatar', '头像', true) on conflict (id) do nothing;
-- 设置 policy
-- insert policy
--   check:
--     1. 操作的必须是 avatar
--     2. 新增的头像必须是当前用户的 \ 路径必须是 avatar/auth.uid().后缀
create policy "avatar_insert_policy" on storage.objects for insert to authenticated with check (
    bucket_id = 'avatar'
    and name ~ (auth.uid()::text || '\.(jpg|jpeg|png|webp)$')
);
-- delete policy
--   using:
--     1. 操作的必须是 avatar
--     2. 只有管理员才能删除
create policy "avatar_delete_policy" on storage.objects for delete to authenticated using (
    bucket_id = 'avatar'
    and is_admin()
);
-- update policy:
--   using:
--     1. 操作的必须是 avatar
--     2. 修改的头像必须是当前用户的 \ 路径必须是 avatar/auth.uid().后缀
--   check:
--     1. 操作的必须是 avatar
--     2. 修改的头像必须是当前用户的 \ 路径必须是 avatar/auth.uid().后缀
create policy "avatar_update_policy" on storage.objects for update to authenticated using (
    bucket_id = 'avatar'
    and name ~ (auth.uid()::text || '\.(jpg|jpeg|png|webp)$')
) with check (
    bucket_id = 'avatar'
    and name ~ (auth.uid()::text || '\.(jpg|jpeg|png|webp)$')
);
-- select policy:
--   using:
--     1. 操作的必须是 avatar
--     2. 自己能查询自己的数据 \ 路径必须是 avatar/auth.uid().后缀
--     3. 管理员可以查询所有数据
create policy "avatar_select_policy" on storage.objects for select to authenticated using (
	bucket_id = 'avatar'
    and (
		is_admin()
		or name in (auth.uid()::text || '.jpg', auth.uid()::text || '.jpeg', auth.uid()::text || '.png', auth.uid()::text || '.webp')
	)
);
-----------------------------------
-- 初始化数据
insert into profile (id, role_id)
select u.id, (select id from "role" where code = 'guest' limit 1)
from auth.users u left join profile p on u.id = p.id
where p.id is null;
commit;

-- 私用触发器函数
create or replace function handle_stock_change()
returns trigger as $$
declare
    v_manufacturer_id uuid;
    v_product_id uuid;
    v_delta decimal(19,4); -- 变化的数量
    v_sign int; -- 符号：入库为 1，出库为 -1
    
    -- 定义临时变量用于安全访问 (解决通用表没有 manufacturer_id 的问题)
    v_record record; 
    v_record_json jsonb;
begin
    -- 1. 确定操作类型和符号
    if tg_argv[0] = 'in' then
        v_sign := 1;
    else
        v_sign := -1;
    end if;

    -- 2. 确定使用的是 NEW 数据还是 OLD 数据
    if (tg_op = 'DELETE') then
        v_record := old;
    else
        v_record := new;
    end if;
    
    -- 转换成 jsonb 以便安全提取字段 (即使字段不存在也不会报错)
    v_record_json := to_jsonb(v_record);

    -- 3. 获取 Manufacturer ID 和 Product ID
    --    分情况处理：合同单据直接取值，通用单据联表查询
    
    -- 【分支 A】: 合同订单相关 (contract_order_stock_in / out)
    if (tg_table_name = 'contract_order_stock_in' or tg_table_name = 'contract_order_stock_out') then
        -- 厂家ID：直接在表里
        v_manufacturer_id := (v_record_json ->> 'manufacturer_id')::uuid;
        -- 产品ID：查 contract_order
        select product_id into v_product_id 
        from contract_order 
        where id = (v_record_json ->> 'contract_order_id')::uuid;

    -- 【分支 B】: 通用入库单 (general_stock_in) -> 关联付款方 (Payer)
    elsif (tg_table_name = 'general_stock_in') then
        select 
            gp.payer_manufacturer_id, -- 入库进付款方的库
            gp.product_id 
        into v_manufacturer_id, v_product_id
        from general_purchase gp 
        where gp.id = (v_record_json ->> 'general_purchase_id')::uuid;

    -- 【分支 C】: 通用出库单 (general_stock_out) -> 关联收款方 (Payee)
    elsif (tg_table_name = 'general_stock_out') then
        select 
            gp.payee_manufacturer_id, -- 出库从收款方的库出
            gp.product_id 
        into v_manufacturer_id, v_product_id
        from general_purchase gp 
        where gp.id = (v_record_json ->> 'general_purchase_id')::uuid;
    end if;

    -- 4. 计算库存变化量 (Delta)
    if (tg_op = 'DELETE') then
        v_delta := (v_record_json ->> 'quantity')::decimal * v_sign * -1;
    elsif (tg_op = 'INSERT') then
        v_delta := (v_record_json ->> 'quantity')::decimal * v_sign;
    elsif (tg_op = 'UPDATE') then
        -- 使用 hstore 检查是否修改了禁止修改的字段
        -- 允许修改的字段: quantity, modifier, modify_time, file_url (如果有), code (如果不允许改code要把code去掉)
        -- 这里反向思考: 只要 Manufacturer ID / Product ID / 关联单据 ID 没变就行
        -- 但是为了严谨, 我们只允许修改数量和文件, 其他一律不准改
        
        declare
            v_allowed_cols text[] := array['quantity', 'modifier', 'modify_time', 'file_url', 'code'];
        begin
            if (hstore(new) - v_allowed_cols) <> (hstore(old) - v_allowed_cols) then
                raise exception '库存单据只允许修改数量和附件! 禁止修改关联信息。';
            end if;
        end;

        v_delta := (new.quantity - old.quantity) * v_sign;
    end if;

    -- 5. 执行库存更新 (Upsert)
    if v_delta is not null and v_delta != 0 and v_manufacturer_id is not null and v_product_id is not null then
        insert into stock (manufacturer_id, product_id, quantity, creator, modifier)
        values (v_manufacturer_id, v_product_id, v_delta, coalesce(auth.uid(), (v_record_json ->> 'creator')::uuid), coalesce(auth.uid(), (v_record_json ->> 'creator')::uuid))
        on conflict (manufacturer_id, product_id) 
        do update set 
            quantity = stock.quantity + excluded.quantity,
            modifier = coalesce(auth.uid(), excluded.modifier),
            modify_time = current_timestamp;
    end if;

    return null;
end;
$$ language plpgsql set search_path = public;

		select 1 from role_todo_rela r where r.role_id = current_role_id() and r.todo_id = t.id
	)
);

-- 创建表
drop table if exists role_todo_rela cascade;
create table role_todo_rela (
    id uuid primary key default uuid_generate_v4(),
    creator uuid not null references profile(id) default auth.uid(),
    modifier uuid not null references profile(id) default auth.uid(),
    create_time timestamptz not null default current_timestamp,
    modify_time timestamptz not null default current_timestamp,
	role_id uuid not null references "role"(id),
	todo_id uuid not null references todo(id),
	constraint uk_role_todo unique (role_id, todo_id)
);

comment on table role_todo_rela is '角色-待办关系';

comment on column role_todo_rela.modify_time is '修改时间';

-- 设置行级安全策略 (RLS)
alter table role_todo_rela enable row level security;
-- 设置 policy
-- insert policy:
--   check:
--     1. 创建人本人
--     2. 未完成状态的待办
create policy "role_todo_rela_insert_policy" on role_todo_rela as r for insert to authenticated with check (
	r.creator = auth.uid()
	and exists (
		select 1 from todo t where t.id = r.todo_id and t.is_completed = false
	)
);
-- delete policy:
--   using:
--     1. 管理员 or 创建人本人
create policy "role_todo_rela_delete_policy" on role_todo_rela as r for delete to authenticated using (
	is_admin()
	or r.creator = auth.uid()
);
-- update policy:
--   using:
--     1. 管理员 or 创建人本人
--   check:
--     1. 修改人本人
create policy "role_todo_rela_update_policy" on role_todo_rela as r for update to authenticated using (
	is_admin()
	or r.creator = auth.uid()
) with check (
	r.modifier = auth.uid()
);
-- select policy:
--   using: 
--     1. 管理员 or 对应的角色
create policy "role_todo_rela_select_policy" on role_todo_rela as r for select to authenticated using (
	is_admin()
	or current_role_id() = r.role_id
);
